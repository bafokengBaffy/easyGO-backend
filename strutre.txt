$p=Get-Location; function t($path,$pre,$last){$item=Get-Item $path -Force; $exclude='node_modules,venv,.venv,__pycache__,.git,.cache,dist,build,.next,.vscode,.idea'; if($exclude -match $item.Name){return}; if($last){$line="$pre└── $($item.Name)";$new=$pre+"    "}else{$line="$pre├── $($item.Name)";$new=$pre+"│   "}; $script:output+= if($script:output){("`n" + $line)}else{$line}; if($item.PSIsContainer){$kids=Get-ChildItem $path -Force|Where-Object{$exclude -notmatch $_.Name}|Sort-Object Name; for($i=0;$i -lt $kids.Count;$i++){t $kids[$i].FullName $new ($i -eq $kids.Count-1)}}}; $output="$(Split-Path $p -Leaf)/`n"; t $p; Clear-Host; Write-Host $output; $output | Out-File -FilePath "filetree.txt" -Encoding UTF8; Write-Host "`nSaved to: filetree.txt" -ForegroundColor Green
